cmake_minimum_required(VERSION 3.16)
project(quantum_gas_transport VERSION 0.3.0 LANGUAGES CXX)
option(QTRANSPORT_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(qtransport_core src/numerics.cpp src/collisions.cpp src/solver.cpp
    src/dynamics.cpp src/thermodynamics.cpp src/workflows.cpp src/figures.cpp)
target_compile_features(qtransport_core PUBLIC cxx_std_17)
target_include_directories(qtransport_core PUBLIC include)
target_include_directories(qtransport_core SYSTEM PUBLIC third_party/eigen third_party)
target_compile_options(qtransport_core PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)
if(QTRANSPORT_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(qtransport_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(qtransport_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(qtransport src/main.cpp tests/tests.cpp)
target_link_libraries(qtransport PRIVATE qtransport_core)
enable_testing()
add_test(NAME physics COMMAND qtransport test --root ${CMAKE_CURRENT_SOURCE_DIR})
install(TARGETS qtransport RUNTIME DESTINATION bin)
