cmake_minimum_required(VERSION 3.16)
project(quantum_gas_transport VERSION 0.4.1 LANGUAGES CXX)
option(QTRANSPORT_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(qtransport_core src/numerics.cpp src/collisions.cpp src/solver.cpp
    src/dynamics.cpp src/thermodynamics.cpp src/workflows.cpp src/figures.cpp)
target_compile_features(qtransport_core PUBLIC cxx_std_17)
target_include_directories(qtransport_core PUBLIC include)
target_include_directories(qtransport_core SYSTEM PUBLIC third_party/eigen third_party)
target_compile_options(qtransport_core PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)
if(QTRANSPORT_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    # Keep Eigen's quadrature-heavy tests practical with sanitizer instrumentation.
    target_compile_options(qtransport_core PUBLIC -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(qtransport_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(qtransport src/main.cpp tests/tests.cpp)
target_link_libraries(qtransport PRIVATE qtransport_core)
enable_testing()
add_test(NAME physics_and_regression COMMAND qtransport test --root ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME cli_help COMMAND qtransport --help)
set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "Commands:")
add_test(NAME cli_transport_smoke COMMAND qtransport transport --statistics classical --basis 1
    --output ${CMAKE_CURRENT_BINARY_DIR}/transport_smoke.json)
install(TARGETS qtransport RUNTIME DESTINATION bin)
add_executable(qtransport_report_export tools/report_export.cpp)
target_compile_features(qtransport_report_export PRIVATE cxx_std_17)
target_include_directories(qtransport_report_export SYSTEM PRIVATE third_party)
